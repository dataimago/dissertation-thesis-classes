\def\thesis@pre@lua{%

    \def\thesis@pre@tag{tagging=on,}%%
    \thesis@pre@is{accessibility.pdfua}{none}\else%
      \thesis@pre@is{accessibility.pdfua}{auto}%%
        \ifx\thesis@pre@ver\@empty\def\thesis@pre@ua{ua-2}\else\def\thesis@pre@ua{ua-1}\fi%
      \else\edef\thesis@pre@ua{\thesis@pre@get{accessibility.pdfua}}\fi\fi%
  \fi%
}
