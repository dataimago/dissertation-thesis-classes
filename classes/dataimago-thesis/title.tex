\thesishaslists{$thesis-derived.has-lof$}{$thesis-derived.has-lot$}
\thesisdate{$thesis-derived.month-year$}{$thesis-derived.year$}
$if(thesis.degree.name)$\thesisdegree{$thesis.degree.name$}$endif$
$if(thesis.degree.abbreviation)$\thesisdegreeabbr{$thesis.degree.abbreviation$}$endif$

$if(thesis.department)$\thesisdepartment{$thesis.department$}$endif$

$if(thesis.irb-protocol)$\thesisirb{$thesis.irb-protocol$}$endif$
$for(thesis.advisors)$
\thesisaddadvisor{$it.name$}{$it.title$}{$it.role$}{$it.affiliation$}
$endfor$
$for(thesis.committee)$
\thesisaddmember{$it.name$}{$it.title$}{$it.role$}{$it.affiliation$}
$endfor$
$for(thesis.previous-degrees)$
\thesisaddprevious{$it$}
$endfor$
$for(thesis.keywords)$
\thesisaddkeyword{$it$}
$endfor$
$if(thesis.draft)$\thesisdraft{true}$endif$
$if(thesis.frontmatter)$\thesisfrontmode{$thesis.frontmatter$}$endif$
